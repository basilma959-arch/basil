    void approveInventory(InventorySession s){int total=allItems(s).size(), counted=0,diffs=0;for(MaterialItem m:allItems(s))if(s.counts.containsKey(m.key()))counted++;for(MaterialItem m:materials){Double q=s.counts.get(m.key());if(q!=null&&Math.abs(q-m.sourceQty)>0.000001)diffs++;}int remaining=total-counted,pct=total==0?0:(int)Math.round(counted*100.0/total);String msg="إجمالي الأصناف: "+total+"\nتم جرده: "+counted+"\nالمتبقي: "+remaining+"\nالإنجاز: "+pct+"%\nالفروقات عن الدفتري: "+diffs+"\n\nسيتم حفظ Snapshot قبل الاعتماد ثم قفل الجرد عن التعديل.";new AlertDialog.Builder(this).setTitle("اعتماد الجرد؟").setMessage(msg).setNegativeButton("إلغاء",null).setPositiveButton("اعتماد",(d,w)->{saveApprovalSnapshot(s);s.approved=true;s.approvedAt=System.currentTimeMillis();saveData();toast("تم اعتماد الجرد وقفل التعديل");showInventories();}).show();}

    void writeInventoryXml(Uri uri,InventorySession s){try(OutputStream out=getContentResolver().openOutputStream(uri)){if(out==null)throw new IOException("تعذر إنشاء الملف");StringBuilder x=new StringBuilder("<?xml version=\"1.0\" encoding=\"utf-8\"?>\n");x.append("<ALBASILStockTake name=\"").append(xml(s.name)).append("\" date=\"").append(xml(s.date)).append("\">\n");for(MaterialItem m:allItems(s)){Double q=s.counts.get(m.key());if(q==null)continue;x.append("  <Item><Code>").append(xml(m.code)).append("</Code><Barcode>").append(xml(m.barcode)).append("</Barcode><Name>").append(xml(m.name)).append("</Name><Qty>").append(fmt(q)).append("</Qty></Item>\n");}x.append("</ALBASILStockTake>");out.write(x.toString().getBytes(StandardCharsets.UTF_8));out.flush();toast("تم تصدير الجرد");}catch(Exception e){toast("تعذر التصدير: "+e.getMessage());}}
