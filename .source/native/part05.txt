    ImportResult parseXml(byte[] bytes)throws Exception{String head=new String(bytes,0,Math.min(bytes.length,6000),StandardCharsets.UTF_8).toLowerCase(Locale.ROOT);if(head.contains("workbook")&&head.contains("spreadsheet"))return parseSpreadsheet(bytes);return parseBasil(bytes);}
    ImportResult parseBasil(byte[] bytes)throws Exception{ImportResult r=new ImportResult();r.type="Basil XML";Document doc=DocumentBuilderFactory.newInstance().newDocumentBuilder().parse(new ByteArrayInputStream(bytes));doc.getDocumentElement().normalize();Map<String,String>groups=new HashMap<>();NodeList all=doc.getElementsByTagName("*");for(int i=0;i<all.getLength();i++){Element e=(Element)all.item(i);String ln=local(e).toLowerCase(Locale.ROOT);if(ln.contains("group")&&!ln.equals("matgroupguid")){String guid=directText(e,"Guid","GroupGuid","ID","Id");String name=directText(e,"GroupName","Name");if(!empty(guid)&&!empty(name))groups.put(guid,name);}}
        for(int i=0;i<all.getLength();i++){Element e=(Element)all.item(i);if(!local(e).equalsIgnoreCase("MaterialItem"))continue;MaterialItem m=new MaterialItem();m.id=first(directText(e,"Guid","MatGuid","MaterialGuid","ID"),"mat_"+UUID.randomUUID());m.code=directText(e,"MatCode","Code");m.name=directText(e,"MatName","Name");String gg=directText(e,"MatGroupGuid","GroupGuid");m.group=first(groups.get(gg),directText(e,"GroupName","Group"),gg);m.barcode=directText(e,"MatBarCode","Barcode");m.barcode2=directText(e,"MatBarCode2","Barcode2");m.barcode3=directText(e,"MatBarCode3","Barcode3");m.unit=directText(e,"MatUnity","Unit");m.unit2=directText(e,"MatUnit2","Unit2");m.unit3=directText(e,"MatUnit3","Unit3");m.factor2=num(directText(e,"MatUnit2Factor","Factor2"),1);m.factor3=num(directText(e,"MatUnit3Factor","Factor3"),1);m.purchasePrice=num(directText(e,"PurchasePrice","LastPurchase","Price"),0);m.sourceQty=num(directText(e,"SourceQuantity","Quantity","Qty"),0);if(!empty(m.name)||!empty(m.code)||!empty(m.barcode))r.items.add(m);}return r;}
