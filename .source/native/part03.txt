    void inventoryMenu(InventorySession s){List<String> opts=new ArrayList<>(Arrays.asList("فتح","تم جرده","لم يُجرد","تصدير XML",s.approved?"إعادة فتح الجرد":"اعتماد الجرد","مراجعة نسخة الاعتماد","حذف الجردة"));new AlertDialog.Builder(this).setTitle(s.name).setItems(opts.toArray(new String[0]),(d,w)->{switch(w){case 0:showInventory(s,"all");break;case 1:showInventory(s,"counted");break;case 2:showInventory(s,"uncounted");break;case 3:exportInventory(s);break;case 4:if(s.approved)reopenInventory(s);else approveInventory(s);break;case 5:reviewApprovalSnapshot(s);break;case 6:deleteInventory(s);break;}}).show();}

    void showInventory(InventorySession s,String mode){currentInventory=s;reportMode=mode;base(s.date+" - "+s.name+(s.approved?"  [معتمد]":""),this::showInventories);LinearLayout b=body();TextView summary=tv("",15,NAVY,true);summary.setBackgroundColor(Color.WHITE);b.addView(summary,new LinearLayout.LayoutParams(-1,dp(58)));addGap(b,7);
        EditText q=input("بحث بالاسم أو جزء منه أو الكود أو الباركود أو المجموعة...");b.addView(q,new LinearLayout.LayoutParams(-1,dp(52)));addGap(b,7);
        LinearLayout tools=new LinearLayout(this);tools.setOrientation(LinearLayout.HORIZONTAL);Button scan=btn("كاميرا الباركود",BLUE);scan.setEnabled(!s.approved);scan.setOnClickListener(v->startScanner());tools.addView(scan,new LinearLayout.LayoutParams(0,dp(52),1));Button manual=btn("+ مادة يدوية",PINK);manual.setEnabled(!s.approved);manual.setOnClickListener(v->manualItemDialog(s,null));tools.addView(manual,new LinearLayout.LayoutParams(0,dp(52),1));b.addView(tools);addGap(b,7);
        TextView modeInfo=tv(mode.equals("counted")?"تقرير: تم جرده":mode.equals("uncounted")?"تقرير: لم يُجرد":"كل المواد",14,Color.DKGRAY,true);b.addView(modeInfo);
        LinearLayout list=new LinearLayout(this);list.setOrientation(LinearLayout.VERTICAL);b.addView(list);Runnable render=()->renderInventory(s,mode,q.getText().toString(),list,summary);q.addTextChangedListener(new SimpleWatcher(render));render.run();
    }
    void renderInventory(InventorySession s,String mode,String query,LinearLayout list,TextView summary){list.removeAllViews();List<MaterialItem> all=allItems(s);int total=all.size(), counted=0;for(MaterialItem m:all)if(s.counts.containsKey(m.key()))counted++;int uncounted=total-counted;int pct=total==0?0:(int)Math.round(counted*100.0/total);summary.setText("إجمالي "+total+" | تم "+counted+" | لم يُجرد "+uncounted+" | الإنجاز "+pct+"%");String q=norm(query);int shown=0;for(MaterialItem m:all){boolean isCount=s.counts.containsKey(m.key());if(mode.equals("counted")&&!isCount)continue;if(mode.equals("uncounted")&&isCount)continue;if(!q.isEmpty()&&!m.matches(q))continue;shown++;addInventoryRow(list,s,m);}if(shown==0)list.addView(tv("لا توجد أصناف مطابقة",16,Color.GRAY,false));}
    void addInventoryRow(LinearLayout list,InventorySession s,MaterialItem m){LinearLayout card=new LinearLayout(this);card.setOrientation(LinearLayout.VERTICAL);card.setPadding(dp(10),dp(7),dp(10),dp(7));card.setBackgroundColor(Color.WHITE);Double val=s.counts.get(m.key());String meta="الكود: "+nz(m.code)+"   الباركود: "+nz(m.barcode)+"   الوحدة: "+nz(m.unit)+"\nالمجموعة: "+nz(m.group)+(m.manual?"   [يدوي]":"");TextView title=tv(m.name+"\n"+meta,15,NAVY,true);card.addView(title);
        LinearLayout qc=new LinearLayout(this);qc.setGravity(Gravity.CENTER);Button minus=btn("−",RED);Button qty=btn(val==null?"—":fmt(val),BLUE);Button plus=btn("+",GREEN);minus.setEnabled(!s.approved);qty.setEnabled(!s.approved);plus.setEnabled(!s.approved);minus.setOnClickListener(v->{double x=s.counts.getOrDefault(m.key(),0.0);s.counts.put(m.key(),Math.max(0,x-1));saveData();showInventory(s,reportMode);});plus.setOnClickListener(v->{double x=s.counts.getOrDefault(m.key(),0.0);s.counts.put(m.key(),x+1);saveData();showInventory(s,reportMode);});qty.setOnClickListener(v->quantityDialog(s,m));qc.addView(minus,new LinearLayout.LayoutParams(0,dp(46),1));qc.addView(qty,new LinearLayout.LayoutParams(0,dp(46),2));qc.addView(plus,new LinearLayout.LayoutParams(0,dp(46),1));card.addView(qc);card.setOnClickListener(v->{if(!s.approved)quantityDialog(s,m);});if(m.manual)card.setOnLongClickListener(v->{if(!s.approved)manualItemActions(s,m);return true;});list.addView(card,new LinearLayout.LayoutParams(-1,-2));addGap(list,6);}
    void quantityDialog(InventorySession s,MaterialItem m){EditText e=input("الكمية");e.setInputType(InputType.TYPE_CLASS_NUMBER|InputType.TYPE_NUMBER_FLAG_DECIMAL);Double v=s.counts.get(m.key());if(v!=null)e.setText(fmt(v));new AlertDialog.Builder(this).setTitle(m.name).setView(e).setNegativeButton("إلغاء",null).setPositiveButton("حفظ",(d,w)->{try{s.counts.put(m.key(),Double.parseDouble(e.getText().toString().trim()));saveData();showInventory(s,reportMode);}catch(Exception ex){toast("أدخل كمية صحيحة");}}).show();}
    void startScanner(){if(currentInventory==null||currentInventory.approved)return;IntentIntegrator x=new IntentIntegrator(this);x.setPrompt("وجّه الكاميرا إلى باركود الصنف");x.setBeepEnabled(true);x.setOrientationLocked(false);x.initiateScan();}
    void handleScan(String code){if(currentInventory==null)return;MaterialItem found=null;for(MaterialItem m:allItems(currentInventory))if(m.matchesExact(code)){found=m;break;}if(found==null){toast("باركود/كود غير مسجل: "+code);return;}quantityDialog(currentInventory,found);}

    void manualItemDialog(InventorySession s,MaterialItem edit){LinearLayout box=new LinearLayout(this);box.setOrientation(LinearLayout.VERTICAL);box.setPadding(dp(18),0,dp(18),0);EditText name=input("الاسم");EditText code=input("الكود");EditText barcode=input("الباركود");EditText unit=input("الوحدة");EditText group=input("المجموعة");EditText qty=input("الكمية");for(EditText e:new EditText[]{name,code,barcode,unit,group,qty})box.addView(e);if(edit!=null){name.setText(edit.name);code.setText(edit.code);barcode.setText(edit.barcode);unit.setText(edit.unit);group.setText(edit.group);Double q=s.counts.get(edit.key());if(q!=null)qty.setText(fmt(q));}
