// Simple calculator: + - * / only
let calcCurrent='0';
let calcStored=null;
let calcPending=null;
let calcWaiting=false;
let calcExpression='';

function calcFmt(n){
  if(!Number.isFinite(n)) return 'خطأ';
  const rounded=Math.round((n+Number.EPSILON)*1e10)/1e10;
  return String(rounded);
}
function calcSymbol(op){ return ({'+':'+','-':'−','*':'×','/':'÷'})[op]||''; }
function calcUpdate(){
  const d=document.getElementById('calcDisplay');
  const o=document.getElementById('calcOperation');
  if(d) d.value=calcCurrent;
  if(o) o.textContent=calcExpression || ((calcStored!==null&&calcPending)?`${calcFmt(calcStored)} ${calcSymbol(calcPending)}`:'');
}
function openCalculator(){
  const x=document.getElementById('calculatorOverlay');
  if(!x) return;
  x.classList.add('open');
  x.setAttribute('aria-hidden','false');
  calcUpdate();
}
function closeCalculator(){
  const x=document.getElementById('calculatorOverlay');
  if(!x) return;
  x.classList.remove('open');
  x.setAttribute('aria-hidden','true');
}
function calcClear(){
  calcCurrent='0'; calcStored=null; calcPending=null; calcWaiting=false; calcExpression=''; calcUpdate();
}
function calcDigit(v){
  if(calcCurrent==='خطأ') calcClear();
  calcExpression='';
  if(calcWaiting){
    calcCurrent=(v==='.')?'0.':v;
    calcWaiting=false;
  }else if(v==='.'){ 
    if(!calcCurrent.includes('.')) calcCurrent+='.';
  }else{
    calcCurrent=(calcCurrent==='0')?v:calcCurrent+v;
  }
  calcUpdate();
}
function calcBackspace(){
  if(calcWaiting || calcCurrent==='خطأ') return;
  calcExpression='';
  calcCurrent=calcCurrent.length>1?calcCurrent.slice(0,-1):'0';
  if(calcCurrent==='' || calcCurrent==='-') calcCurrent='0';
  calcUpdate();
}
function calcCompute(a,b,op){
  if(op==='+') return a+b;
  if(op==='-') return a-b;
  if(op==='*') return a*b;
  if(op==='/') return b===0?NaN:a/b;
  return b;
}
function calcOperator(op){
  if(calcCurrent==='خطأ') calcClear();
  const input=Number(calcCurrent);
  calcExpression='';
  if(calcStored===null){
    calcStored=input;
  }else if(calcPending && !calcWaiting){
    const r=calcCompute(calcStored,input,calcPending);
    if(!Number.isFinite(r)){
      calcCurrent='خطأ'; calcStored=null; calcPending=null; calcWaiting=true; calcUpdate(); return;
    }
    calcStored=r;
    calcCurrent=calcFmt(r);
  }
  calcPending=op;
  calcWaiting=true;
  calcUpdate();
}
function calcEquals(){
  if(calcStored===null || !calcPending || calcCurrent==='خطأ') return;
  const b=Number(calcCurrent);
  const a=calcStored;
  const op=calcPending;
  const r=calcCompute(a,b,op);
  if(!Number.isFinite(r)){
    calcCurrent='خطأ';
    calcExpression=`${calcFmt(a)} ${calcSymbol(op)} ${calcFmt(b)} =`;
  }else{
    calcCurrent=calcFmt(r);
    calcExpression=`${calcFmt(a)} ${calcSymbol(op)} ${calcFmt(b)} =`;
  }
  calcStored=null;
  calcPending=null;
  calcWaiting=true;
  calcUpdate();
}
